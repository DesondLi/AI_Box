```python
"""
测试辅助工具和通用 Mock 类
"""
from typing import List, Any
from kernel.interfaces import IPlugin, IKernelContext


class LifecycleTracker:
    """跟踪插件生命周期调用"""
    def __init__(self):
        self.events: List[str] = []

    def record(self, event: str):
        self.events.append(event)


class MockPlugin(IPlugin):
    """通用 Mock 插件

    支持：
    - 自定义依赖列表
    - 在指定阶段抛出异常（install/start/stop）
    - 记录生命周期调用
    """
    def __init__(self, plugin_id: str, deps: list = None, fail_at: str = None):
        self._plugin_id = plugin_id
        self._deps = deps or []
        self.fail_at = fail_at  # 'install' / 'start' / 'stop' / None
        self.tracker = LifecycleTracker()
        self.context = None

    @property
    def plugin_id(self) -> str:
        return self._plugin_id

    @property
    def version(self) -> str:
        return "1.0.0"

    @property
    def dependencies(self) -> list:
        return self._deps

    async def install(self, context: IKernelContext):
        self.tracker.record("install")
        self.context = context
        if self.fail_at == "install":
            raise RuntimeError(f"Install failed for {self._plugin_id}")

    async def start(self):
        self.tracker.record("start")
        if self.fail_at == "start":
            raise RuntimeError(f"Start failed for {self._plugin_id}")

    async def stop(self):
        self.tracker.record("stop")
        if self.fail_at == "stop":
            raise RuntimeError(f"Stop failed for {self._plugin_id}")

    async def uninstall(self):
        self.tracker.record("uninstall")


class CallbackTracker:
    """跟踪回调函数调用次数和参数"""
    def __init__(self):
        self.calls: List[tuple] = []

    def __call__(self, *args, **kwargs):
        self.calls.append((args, kwargs))

    @property
    def call_count(self) -> int:
        return len(self.calls)

    @property
    def last_call(self) -> tuple:
        return self.calls[-1] if self.calls else None
```

    def test_watch_receives_initial_set(self, temp_config):
        """监听器收到首次 set 通知"""
        tracker = CallbackTracker()
        temp_config.watch("test/*", tracker)

        temp_config.set("test/key", "value")

        assert tracker.call_count == 1
        assert tracker.last_call[0] == ("test/key", "value")

    def test_watch_receives_update(self, temp_config):
        """监听器收到值变更通知"""
        temp_config.set("test/key", "v1")

        tracker = CallbackTracker()
        temp_config.watch("test/*", tracker)

        temp_config.set("test/key", "v2")
        assert tracker.call_count == 1

    def test_watch_no_trigger_on_same_value(self, temp_config):
        """相同值不触发通知"""
        temp_config.set("test/key", "v1")

        tracker = CallbackTracker()
        temp_config.watch("test/*", tracker)

        temp_config.set("test/key", "v1")  # 相同值
        assert tracker.call_count == 0

    def test_watch_pattern_matching(self, temp_config):
        """fnmatch 模式匹配正确"""
        tracker = CallbackTracker()
        temp_config.watch("sensor.*", tracker)

        temp_config.set("sensor.temp", 25)
        temp_config.set("sensor.humidity", 60)
        temp_config.set("actuator.light", "on")  # 不匹配

        assert tracker.call_count == 2

    def test_unwatch_stops_callbacks(self, temp_config):
        """取消监听后不再收到通知"""
        tracker = CallbackTracker()
        watch_id = temp_config.watch("test/*", tracker)

        temp_config.set("test/a", 1)
        assert tracker.call_count == 1

        temp_config.unwatch(watch_id)

        temp_config.set("test/b", 2)
        assert tracker.call_count == 1  # 没有新增调用

    def test_unwatch_nonexistent_is_safe(self, temp_config):
        """取消不存在的 watch_id 不会报错"""
        temp_config.unwatch("nonexistent-watch-id")  # 不应抛出异常

    def test_watcher_exception_silently_ignored(self, temp_config):
        """监听器抛出异常不影响主流程"""
        def bad_callback(key, value):
            raise RuntimeError("Callback failed")

        temp_config.watch("test/*", bad_callback)
        temp_config.set("test/key", "value")  # 不应抛出异常

    def test_multiple_watchers_independent(self, temp_config):
        """多个监听器独立工作"""
        tracker1 = CallbackTracker()
        tracker2 = CallbackTracker()

        temp_config.watch("a.*", tracker1)
        temp_config.watch("b.*", tracker2)

        temp_config.set("a.x", 1)
        temp_config.set("b.y", 2)

        assert tracker1.call_count == 1
        assert tracker2.call_count == 1

错误类型系统测试

测试目标：

class TestResultType:
    """Result 类型测试"""

    def test_ok_result(self):
        """成功结果"""
        result = Ok(42)
        assert result.is_ok()
        assert not result.is_err()
        assert result.unwrap() == 42

    def test_err_result(self):
        """错误结果"""
        err = PluginError("E001", "Failed")
        result = Err(err)
        assert not result.is_ok()
        assert result.is_err()
        assert result.unwrap_err() is err

    def test_ok_unwrap_err_raises(self):
        """Ok 结果调用 unwrap_err 抛出异常"""
        result = Ok(42)
        with pytest.raises(RuntimeError, match="Called unwrap_err on Ok"):
            result.unwrap_err()

    def test_err_unwrap_raises(self):
        """Err 结果调用 unwrap 抛出异常"""
        err = PluginError("E001", "Failed")
        result = Err(err)
        with pytest.raises(PluginError):
            result.unwrap()

        """map 转换 Ok 值"""
        result = Ok(21)
        mapped = result.map(lambda x: x * 2)
        assert mapped.is_ok()
        assert mapped.unwrap() == 42

    def test_result_map_on_err(self):
        """map 在 Err 上不执行"""
        err = PluginError("E001", "Failed")
        result = Err(err)
        called = []
        mapped = result.map(lambda x: called.append(x))
        assert mapped.is_err()
        assert len(called) == 0  # 函数未执行

    def test_result_map_err_on_err(self):
        """map_err 转换 Err 值"""
        original = PluginError("E001", "Original")
        result = Err(original)

        def wrap_err(e):
            return PluginError("E999", f"Wrapped: {e.message}")

        mapped = result.map_err(wrap_err)
        assert mapped.is_err()
        assert mapped.unwrap_err().code == "E999"

    def test_result_map_err_on_ok(self):
        """map_err 在 Ok 上不执行"""
        result = Ok(42)
        called = []
        mapped = result.map_err(lambda e: called.append(e))
        assert mapped.is_ok()
        assert len(called) == 0

    def test_result_and_then_on_ok(self):
        """and_then 链式调用 Ok"""
        result = Ok(20)
        chained = result.and_then(lambda x: Ok(x + 22))
        assert chained.is_ok()
        assert chained.unwrap() == 42

    def test_result_and_then_on_err(self):
        """and_then 在 Err 上短路"""
        err = PluginError("E001", "Failed")
        result = Err(err)
        called = []
        chained = result.and_then(lambda x: called.append(x) or Ok(x))
        assert chained.is_err()
        assert len(called) == 0

    def test_result_or_else_on_ok(self):
        """or_else 在 Ok 上不执行"""
        result = Ok(42)
        called = []
        recovered = result.or_else(lambda e: called.append(e) or Ok(0))
        assert recovered.is_ok()
        assert recovered.unwrap() == 42
        assert len(called) == 0

    def test_result_or_else_on_err_recovers(self):
        """or_else 从 Err 恢复"""
        err = PluginError("E001", "Recoverable")
        result = Err(err)
        recovered = result.or_else(lambda e: Ok(f"Recovered: {e.message}"))
        assert recovered.is_ok()
        assert recovered.unwrap() == "Recovered: Recoverable"

    def test_ok_convenience_function(self):
        """Ok() 快捷函数返回 Result 对象"""
        result = Ok("value")
        assert isinstance(result, Result)
        assert result.is_ok()

    def test_err_convenience_function(self):
        """Err() 快捷函数返回 Result 对象"""
        err = PluginError("E001", "Error")
        result = Err(err)
        assert isinstance(result, Result)
        assert result.is_err()
